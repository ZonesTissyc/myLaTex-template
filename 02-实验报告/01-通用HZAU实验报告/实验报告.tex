\begin{reportcode}[language=C++]{OpenMP 并行计算示例}
#include <iostream>
int main() {
    const int steps = 1000000; double sum = 0.0;
    // Sum the areas of rectangles in parallel.
    #pragma omp parallel for reduction(+:sum)
    for (int i = 0; i < steps; ++i) {
        double x = (i + 0.5) / steps;
        sum += 4.0 / (1.0 + x * x);
    }
    std::cout << "pi = " << sum / steps << std::endl;
}
\end{reportcode}
